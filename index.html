/* ==================================================
   対戦AI 日本語表示
================================================== */

function formatBattleJapanese(data) {

  if (!data) {
    return "AIの結果がありません。";
  }

  const best =
    data.bestAction;

  const candidates =
    data.candidates || [];

  const input =
    data.input || {};

  let text = "";

  text +=
    "🧠 対戦AIのおすすめ\n";

  text +=
    "━━━━━━━━━━━━━━━━\n\n";


  /* =========================
     対面
  ========================= */

  if (
    input.me &&
    input.opponent
  ) {

    text +=
      "⚔️ 現在の対面\n";

    text +=
      "自分： " +
      input.me +
      "\n";

    text +=
      "相手： " +
      input.opponent +
      "\n\n";

  }


  /* =========================
     おすすめ
  ========================= */

  if (best) {

    const action =
      best.action || [];


    if (
      action[0] === "switch"
    ) {

      text +=
        "🔄 おすすめ：" +
        action[1] +
        " に交代\n\n";

    } else {

      const move =
        best.move ||
        action[1] ||
        "不明";

      text +=
        "⚔️ おすすめ技：" +
        move +
        "\n\n";

    }


    text +=
      "【理由】\n";


    if (
      Array.isArray(
        best.reason
      ) &&
      best.reason.length
    ) {

      best.reason.forEach(
        function(reason) {

          text +=
            "・" +
            translateBattleReason(
              reason
            ) +
            "\n";

        }
      );

    }


    if (
      best.estimatedDamage != null
    ) {

      text +=
        "\n【予想ダメージ】\n";

      text +=
        "・相手に与える：約" +
        best.estimatedDamage +
        "\n";

    }


    if (
      best.incomingDamage != null
    ) {

      text +=
        "・自分が受ける：約" +
        best.incomingDamage +
        "\n";

    }

  }


  /* =========================
     候補
  ========================= */

  text +=
    "\n━━━━━━━━━━━━━━━━\n";

  text +=
    "📋 その他の候補\n\n";


  candidates.forEach(
    function(c, index) {

      const action =
        c.action || [];

      text +=
        (index + 1) +
        ". ";


      if (
        action[0] === "switch"
      ) {

        text +=
          "🔄 " +
          action[1] +
          " に交代";

      } else {

        text +=
          "⚔️ " +
          (
            c.move ||
            action[1] ||
            "不明"
          );

      }


      if (
        c.score != null
      ) {

        text +=
          "　（評価：" +
          c.score +
          "）";

      }


      text += "\n";


      if (
        Array.isArray(c.reason) &&
        c.reason.length
      ) {

        text +=
          "   理由：";

        text +=
          c.reason
            .map(
              translateBattleReason
            )
            .join("、");

        text += "\n";

      }


      if (
        c.estimatedDamage != null &&
        c.estimatedDamage > 0
      ) {

        text +=
          "   与ダメージ：約" +
          c.estimatedDamage +
          "\n";

      }


      if (
        c.incomingDamage != null &&
        c.incomingDamage > 0
      ) {

        text +=
          "   被ダメージ：約" +
          c.incomingDamage +
          "\n";

      }


      text += "\n";

    }
  );


  return text;

}

   理由を日本語にする
================================================== */

) {

  const map = {

    "無効":
      "相手には無効",

    "いまひとつ":

      "状態異常を狙える",

    "Protect":
      "Protectで相手の攻撃を防げる",

    "交代可能":
      "攻撃後に交代できる",

    "相手が先に動く可能性":
      "相手が先に動く可能性がある",

    "こちらが先に動く可能性":
      "こちらが先に動く可能性がある",

    "素早さは同じ":
      "素早さは同じ",

    "交代先から効果抜群を狙える":
      "交代先から効果抜群を狙える",

    "交代先が相手より遅い":
      "交代先のほうが相手より遅い",

    "交代先が相手より速い":
      "交代先のほうが相手より速い",

    "相手の攻撃が弱点":
      "相手の攻撃が弱点になる",

    "相手に等倍打点":
      "相手に等倍のダメージを与えられる",

    "相手の攻撃を軽減しやすい":
